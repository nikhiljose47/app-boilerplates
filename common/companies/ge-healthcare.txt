Bloc state -
class DynamicWidgetState {
  final String value;
  const DynamicWidgetState(this.value);
}


Bloc event - 
abstract class DynamicWidgetEvent {}

class StreamValueUpdated extends DynamicWidgetEvent {
  final String value;

  StreamValueUpdated(this.value);
}


Bloc-  //Bloc receives the stream through constructor.
class DynamicWidgetBloc
    extends Bloc<DynamicWidgetEvent, DynamicWidgetState> {

  late final StreamSubscription _subscription;

  DynamicWidgetBloc(Stream<String> dataStream)
      : super(const DynamicWidgetState("")) {

    on<StreamValueUpdated>((event, emit) {
      emit(DynamicWidgetState(event.value));
    });

    _subscription = dataStream.listen((value) {
      add(StreamValueUpdated(value));
    });
  }

  @override
  Future<void> close() {
    _subscription.cancel();
    return super.close();
  }
}

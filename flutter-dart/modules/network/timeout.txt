Future<Response> fetchWithRetry(Dio dio, String url) async {
  int retries = 0;

  while (retries < 3) {
    try {
      return await dio.get(url);
    } catch (e) {
      retries++;
      if (retries == 3) rethrow;
      await Future.delayed(Duration(seconds: 1));
    }
  }

  throw Exception("Failed after retries");
}
